import { createHash } from "node:crypto";
import { readFileSync } from "node:fs";
import { sidecarPin } from "./g2-r6-pin-helper.js";
const GLB = "/repo/apps/ui-xr/public/generated-humanoids/mpfb-test.glb";
const live = createHash("sha256").update(readFileSync(GLB)).digest("hex");
const want = sidecarPin();
if (live !== want) throw new Error("moved");
