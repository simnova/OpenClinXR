import { createHash } from "node:crypto";
import { readFileSync } from "node:fs";
const GLB = "/repo/apps/ui-xr/public/generated-humanoids/mpfb-test.glb";
const PIN = "1414141414141414141414141414141414141414141414141414141414141414";
export function check(): void {
  const live = createHash("sha256").update(readFileSync(GLB)).digest("hex");
  if (live !== PIN) throw new Error("moved");
}
