import { createHash } from "node:crypto";
import * as fs from "node:fs";
const PIN = "ffffffffffffffffffffffffffffffffffffffffffffffffffffffffffffffff";
async function main(): Promise<void> {
  const live = createHash("sha256").update(await fs.promises.readFile("a.glb")).digest("hex");
  if (live !== PIN) throw new Error("moved");
}
void main();
