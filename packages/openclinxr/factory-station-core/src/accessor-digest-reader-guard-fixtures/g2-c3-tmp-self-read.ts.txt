import { createHash } from "node:crypto";
import { mkdtempSync, readFileSync, writeFileSync } from "node:fs";
import { tmpdir } from "node:os";
import { join } from "node:path";
const PIN = "ffffffffffffffffffffffffffffffffffffffffffffffffffffffffffffffff";
const dir = mkdtempSync(join(tmpdir(), "g2c3-"));
const p = join(dir, "x.glb");
writeFileSync(p, Buffer.from("fixture"));
const live = createHash("sha256").update(readFileSync(p)).digest("hex");
if (live !== PIN) throw new Error("moved");
